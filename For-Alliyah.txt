    function sparkle() {

        const s = document.createElement("div");
        s.className = "sparkle";
        s.textContent = "💙";
        s.style.left = Math.random() * window.innerWidth + "px";
        s.style.top = Math.random() * window.innerHeight + "px";
        document.body.appendChild(s);
